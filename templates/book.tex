% examples/book/templates/book.tex
%
% Example book template for the skelepress toolkit. This is NOT the toolkit
% default (templates/default.tex): it is a self-contained demonstration of a
% \documentclass{book} publication, written for examples/book/ and meant to be
% copied and adapted by consumers.
%
% It covers: page geometry, running headers/footers, a title page driven by
% the document.* variables, the table of contents behind the "toc" condition,
% toolkit style code blocks (fancyvrb + fvextra + mdframed with Highlighting
% support through the "highlighting-macros" variable), graphics,
% longtable/booktabs tables and hyperref with hidelinks.
%
% Engine detection mirrors templates/default.tex: XeLaTeX/LuaLaTeX take the
% fontspec branch, pdfLaTeX keeps T1 fontenc + lmodern. This example template
% deliberately does NOT include the D14 symbol fallback for the six checklist
% glyphs; content that uses those symbols is out of scope here.

\documentclass[11pt,openany]{book}

% ---- Engine detection (mirrors templates/default.tex) ----
\newif\ifxetexorlua
\ifdefined\luatexversion
    \xetexorluatrue
\else
    \ifdefined\XeTeXversion
        \xetexorluatrue
    \else
        \xetexorluafalse
    \fi
\fi

\ifxetexorlua
    \usepackage{fontspec}
    % Same fallback chain as the default template: TeX Gyre Pagella ->
    % Times New Roman -> Latin Modern Roman.
    \IfFontExistsTF{TeX Gyre Pagella}{
        \setmainfont{TeX Gyre Pagella}
    }{
        \IfFontExistsTF{Times New Roman}{
            \setmainfont{Times New Roman}
        }{
            \setmainfont{Latin Modern Roman}
        }
    }
\else
    \usepackage[T1]{fontenc}
    \usepackage[utf8]{inputenc}
    \usepackage{lmodern}
\fi

% ---- Language ----
\usepackage[$if(babel-lang)$$babel-lang$$else$english$endif$]{babel}

% ---- Page geometry ----
\usepackage[a4paper,top=2.5cm,bottom=2.5cm,inner=2.5cm,outer=2cm]{geometry}
\setlength{\emergencystretch}{3em}

% ---- Headers/footers ----
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\chaptermark}[1]{\markboth{#1}{}}
\renewcommand{\sectionmark}[1]{\markright{#1}}
\fancyhead[LE]{\small\itshape\nouppercase{\leftmark}}
\fancyhead[RO]{\small\itshape\nouppercase{\rightmark}}
\fancyfoot[LE,RO]{\small\thepage}
\renewcommand{\headrulewidth}{0.4pt}

% ---- Graphics ----
\usepackage{graphicx}
\usepackage{etoolbox}
% Only downscale images wider than the text block; never upscale small ones.
\makeatletter
\def\maxwidth{\ifdim\Gin@nat@width>\linewidth 0.90\linewidth\else\Gin@nat@width\fi}
\def\maxheight{\ifdim\Gin@nat@height>0.75\textheight 0.75\textheight\else\Gin@nat@height\fi}
\makeatother
\setkeys{Gin}{width=\maxwidth, height=\maxheight, keepaspectratio}

% ---- Colors ----
\usepackage{xcolor}
\definecolor{codegray}{gray}{0.96}

% ---- Code blocks: toolkit look (fancyvrb + fvextra + mdframed) ----
\usepackage{fancyvrb}
\usepackage{fvextra} % adds the 'breaklines' key for Verbatim
\RecustomVerbatimEnvironment{verbatim}{Verbatim}{}
\usepackage{framed}
\fvset{fontsize=\small}
\usepackage{mdframed}

\BeforeBeginEnvironment{Highlighting}{%
    \begin{mdframed}[
    backgroundcolor=codegray,
    roundcorner=3pt,
    leftmargin=0pt,
    rightmargin=0pt,
    innerleftmargin=6pt,
    innerrightmargin=6pt,
    innertopmargin=4pt,
    innerbottommargin=4pt,
    linewidth=0pt
    ]}
\AfterEndEnvironment{Highlighting}{\end{mdframed}}
% Pandoc highlighting macros (KeywordTok, DataTypeTok, ...).
$if(highlighting-macros)$
$highlighting-macros$
$endif$
% Guarded fallbacks, in case Pandoc did not emit the environments.
\makeatletter
\@ifundefined{Shaded}{
    \definecolor{shadecolor}{gray}{0.95}
    \newenvironment{Shaded}{\begin{snugshade}}{\end{snugshade}}
}{}
\@ifundefined{Highlighting}{
    \DefineVerbatimEnvironment{Highlighting}{Verbatim}{commandchars=\\\{\}}
}{}
\makeatother
\fvset{breaklines=true,breakanywhere=false,breakafter={/.,=}}

% ---- Tables: longtable + booktabs ----
\usepackage{longtable,booktabs,array}
\usepackage{calc}

% ---- Pandoc compatibility helpers ----
\providecommand{\tightlist}{%
    \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\providecommand{\pandocbounded}[1]{%
    \begin{minipage}{\linewidth}#1\end{minipage}}

% ---- Book-level headings -------------------------------------------------
% Top-level Markdown headings become \chapter through the configuration: the
% renderer passes --top-level-division=chapter (pandoc.top_level_division in
% skelepress.config.yaml, which the bookkit adapter emits for every book project), so
% the template redefines no heading command. Numbering follows from the class
% automatically: chapters print 1, 2, ... and, after \appendix in the content,
% A, B, ...

% ---- Links ----
\usepackage[hidelinks]{hyperref}

% ---- Contents entries the adapter can hide ----
% The contents list every chapter by default. The bookkit adapter emits
% \BookChaptersUnlisted around an entry declared with toc: false and
% \BookChaptersListed after it, so a dedication can stay out of the contents
% while a prologue stays in.
%
% It works by swallowing the contents entries while the flag is off, not by
% changing the chapter command: the class itself calls \chapter* for the table
% of contents, and redefining \chapter breaks that. A chapter written while the
% flag is off keeps its (front-matter) numbering rules and simply is not listed.
%
% This must come after hyperref loads: hyperref redefines \addcontentsline, and
% a definition placed before it would be silently replaced.
\makeatletter
\newif\ifBookChapterListed
\BookChapterListedtrue
\let\BookAddContentsLine\addcontentsline
\renewcommand\addcontentsline[3]{%
  \ifBookChapterListed\BookAddContentsLine{#1}{#2}{#3}\fi}
\newcommand\BookChaptersListed{\BookChapterListedtrue}
\newcommand\BookChaptersUnlisted{\BookChapterListedfalse}
\makeatother

% ---- Title page ----
\newcommand{\doTitlePage}{%
\begin{titlepage}
\centering
\vspace*{2cm}
{\Huge\bfseries $if(title)$$title$$endif$\par}
$if(subtitle)$
    \vspace{0.6cm}{\Large $subtitle$\par}
$endif$
$if(author)$
    \vspace{1.2cm}{\large $author$\par}
$endif$
\vfill
$if(publisher)$
    {\large $publisher$\par}
$endif$
$if(date)$
    \vspace{0.2cm}{\large $date$\par}
$endif$
$if(rights)$
    \vspace{0.6cm}{\small $rights$\par}
$endif$
\end{titlepage}
\pagenumbering{gobble}
\clearpage
\pagenumbering{arabic}
}

% ---- Book-level mode commands --------------------------------------------
% This template declares openany, but TeX Live 2026's book.cls calls
% \cleardoublepage unconditionally in \frontmatter and \mainmatter, which
% forces recto starts and inserts blank pages. Redefine them to respect the
% openany choice: one \clearpage, then the mode switch.
\makeatletter
\renewcommand\frontmatter{\clearpage\@mainmatterfalse\pagenumbering{roman}}
\renewcommand\mainmatter{\clearpage\@mainmattertrue\pagenumbering{arabic}}
\makeatother

% ---- Document ----
\begin{document}

$if(cover-image)$
% ---- Image cover page ----
% A declared cover image is not a title page: it precedes the typographic
% title page and prints even when the titlepage switch is false. The
% newgeometry/restoregeometry pair lets the image use the whole sheet,
% whatever the body margins are.
\newgeometry{margin=0pt}
\thispagestyle{empty}
\noindent\includegraphics[width=\paperwidth,height=\paperheight,keepaspectratio]{$cover-image$}
\restoregeometry
$endif$

$if(titlepage)$
    \doTitlePage
$endif$

% The template opens the front matter, so the table of contents and the front
% files share one continuous roman sequence. The first chapter opens with
% \mainmatter and the first back-matter file with \appendix; see
% examples/book/README.md.
\frontmatter

$if(toc)$
\tableofcontents
\clearpage
$endif$

$body$

\end{document}
